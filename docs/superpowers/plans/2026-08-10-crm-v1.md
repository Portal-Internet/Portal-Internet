# CRM Portal Internet v1 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Substituir a coleta manual de cadastro do script de vendas por um link de contratação gerado no CRM, um formulário online no site e uma lista de clientes com status.

**Architecture:** Três peças e nenhum servidor novo. O site institucional (SPA estática em Apache) ganha a rota `/contratar/:token` e fala com o Supabase por `fetch` cru em duas funções RPC `security definer`. O CRM é uma SPA Vite nova, hospedada na Vercel, que usa o SDK do Supabase para sessão e leitura. O RLS nega tudo ao papel `anon`, e todo o poder de escrita do público está confinado a uma função que valida um token de uso único dentro de uma transação.

**Tech Stack:** Postgres/Supabase (Auth, RLS, pgTAP), React 19 + TypeScript strict, Vite, react-router-dom, CSS Modules, Vitest, Vercel.

**Spec:** `docs/superpowers/specs/2026-08-10-crm-contratacao-design.md`

## Global Constraints

- **Dois repositórios.** Caminhos prefixados com `crm:` são relativos a `C:\Projetos\portalinternet-crm` (criado na Task 1). Caminhos com `site:` são relativos a `C:\Projetos\PortalInternet` (existente).
- **O site nunca instala `@supabase/supabase-js`.** Duas chamadas RPC via `fetch`. O SDK custaria ~35 kB gzip num bundle afinado para 98 de Lighthouse.
- **Nenhum caminho de asset absoluto no JSX do site.** Usar `asset()` de `@/shared/lib/assets` — o build em subpasta quebra strings escritas à mão.
- **A chave `service_role` nunca sai da máquina do administrador.** Não entra na Vercel, não entra no site, não entra em `.env` versionado.
- **TypeScript strict**, named exports, um componente por arquivo, barrel `index.ts` por pasta de feature/página, CSS Modules colocado ao lado do componente, alias `@/`.
- **Nunca hardcodar cor ou espaçamento** nos módulos CSS — usar os custom properties `--pi-*`.
- **Expiração do link: 72 horas.** Uso único.
- **Status inicial do cliente: `aguardando_instalacao`** ("plano contratado, instalação pendente").
- **`plano_id` é `text`** e espelha os ids de `site:src/shared/data/plans.ts`: `home-office`, `gamer`, `streaming`, `empresarial`, `business`, `executivo`, `dedicado-100`, `dedicado-200`, `dedicado-500`, `dedicado-1000`.
- **Mensagens de commit em português**, prefixo convencional (`feat:`, `test:`, `chore:`, `docs:`).

```sql
create extension if not exists pgcrypto;

create type link_status    as enum ('pendente', 'preenchido', 'expirado', 'cancelado');
create type cliente_status as enum ('aguardando_instalacao', 'instalacao_agendada', 'instalado', 'cancelado');
create type origem_lead    as enum ('link_vendedor', 'site', 'manual');
create type pessoa_tipo    as enum ('fisica', 'juridica');

create table contract_links (
  id               uuid primary key default gen_random_uuid(),
  token            text not null unique,
  status           link_status not null default 'pendente',
  prefill_nome     text,
  prefill_telefone text,
  plano_id         text not null,
  gerado_por       uuid references auth.users (id),
  expira_em        timestamptz not null default now() + interval '72 hours',
  criado_em        timestamptz not null default now(),
  consumido_em     timestamptz
);

create index contract_links_status_idx on contract_links (status);

create table clientes (
  id               uuid primary key default gen_random_uuid(),
  nome_completo    text not null,
  tipo             pessoa_tipo not null,
  documento        text not null,
  rg               text,
  email            text not null,
  data_nascimento  date,
  telefone         text not null,
  cep              text not null,
  logradouro       text not null,
  numero           text not null,
  complemento      text,
  bairro           text not null,
  cidade           text not null,
  uf               char(2) not null,
  plano_id         text not null,
  status           cliente_status not null default 'aguardando_instalacao',
  origem           origem_lead not null default 'link_vendedor',
  link_id          uuid references contract_links (id),
  criado_por       uuid references auth.users (id),
  consentimento_em timestamptz,
  observacoes      text,
  criado_em        timestamptz not null default now(),
  atualizado_em    timestamptz not null default now()
);

create index clientes_status_idx    on clientes (status);
create index clientes_criado_em_idx on clientes (criado_em desc);
create index clientes_documento_idx on clientes (documento);

create table cliente_historico (
  id         uuid primary key default gen_random_uuid(),
  cliente_id uuid not null references clientes (id) on delete cascade,
  de         cliente_status,
  para       cliente_status not null,
  nota       text,
  autor      uuid references auth.users (id),
  criado_em  timestamptz not null default now()
);

create index cliente_historico_cliente_idx on cliente_historico (cliente_id, criado_em desc);

create or replace function set_atualizado_em()
returns trigger language plpgsql as $$
begin
  new.atualizado_em := now();
  return new;
end;
$$;

create trigger clientes_atualizado_em
  before update on clientes
  for each row execute function set_atualizado_em();

-- O historico e responsabilidade do banco, nao do front: assim nenhuma
-- mudanca de status escapa por esquecimento de quem chamou o update.
create or replace function log_cliente_status()
returns trigger language plpgsql as $$
begin
  if tg_op = 'INSERT' then
    insert into cliente_historico (cliente_id, de, para, autor)
    values (new.id, null, new.status, new.criado_por);
  elsif new.status is distinct from old.status then
    insert into cliente_historico (cliente_id, de, para, autor)
    values (new.id, old.status, new.status, auth.uid());
  end if;
  return new;
end;
$$;

create trigger clientes_log_status
  after insert or update of status on clientes
  for each row execute function log_cliente_status();
```

```sql
create or replace function validar_cpf(doc text)
returns boolean
language plpgsql immutable
as $$
declare
  d     text := regexp_replace(coalesce(doc, ''), '\D', '', 'g');
  soma  int;
  resto int;
  i     int;
begin
  if length(d) <> 11 then
    return false;
  end if;

  if d ~ '^(\d)\1{10}$' then
    return false;
  end if;

  soma := 0;
  for i in 1..9 loop
    soma := soma + substr(d, i, 1)::int * (11 - i);
  end loop;
  resto := (soma * 10) % 11;
  if resto = 10 then
    resto := 0;
  end if;
  if resto <> substr(d, 10, 1)::int then
    return false;
  end if;

  soma := 0;
  for i in 1..10 loop
    soma := soma + substr(d, i, 1)::int * (12 - i);
  end loop;
  resto := (soma * 10) % 11;
  if resto = 10 then
    resto := 0;
  end if;

  return resto = substr(d, 11, 1)::int;
end;
$$;

create or replace function validar_cnpj(doc text)
returns boolean
language plpgsql immutable
as $$
declare
  d      text  := regexp_replace(coalesce(doc, ''), '\D', '', 'g');
  pesos1 int[] := array[5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2];
  pesos2 int[] := array[6, 5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2];
  soma   int;
  resto  int;
  i      int;
begin
  if length(d) <> 14 then
    return false;
  end if;

  if d ~ '^(\d)\1{13}$' then
    return false;
  end if;

  soma := 0;
  for i in 1..12 loop
    soma := soma + substr(d, i, 1)::int * pesos1[i];
  end loop;
  resto := soma % 11;
  resto := case when resto < 2 then 0 else 11 - resto end;
  if resto <> substr(d, 13, 1)::int then
    return false;
  end if;

  soma := 0;
  for i in 1..13 loop
    soma := soma + substr(d, i, 1)::int * pesos2[i];
  end loop;
  resto := soma % 11;
  resto := case when resto < 2 then 0 else 11 - resto end;

  return resto = substr(d, 14, 1)::int;
end;
```

```sql
-- Toda escrita publica passa por aqui. As funcoes sao security definer com
-- search_path fixo: o papel anon nao tem acesso nenhum as tabelas, so
-- permissao de executar duas destas tres funcoes.

create or replace function create_contract_link(
  p_nome     text,
  p_telefone text,
  p_plano_id text
)
returns text
language plpgsql security definer set search_path = public
as $$
declare
  v_token text;
begin
  if auth.uid() is null then
    raise exception 'nao autenticado' using errcode = '42501';
  end if;

  -- 16 bytes em base64url, sem padding: 22 caracteres seguros em URL.
  v_token := rtrim(
    translate(encode(gen_random_bytes(16), 'base64'), '+/', '-_'),
    '='
  );

  insert into contract_links (token, prefill_nome, prefill_telefone, plano_id, gerado_por)
  values (
    v_token,
    nullif(trim(coalesce(p_nome, '')), ''),
    nullif(regexp_replace(coalesce(p_telefone, ''), '\D', '', 'g'), ''),
    p_plano_id,
    auth.uid()
  );

  return v_token;
end;
$$;

revoke all on function create_contract_link(text, text, text) from public, anon;
grant execute on function create_contract_link(text, text, text) to authenticated;

create or replace function get_contract_link(p_token text)
returns jsonb
language plpgsql security definer set search_path = public
as $$
declare
  v contract_links%rowtype;
begin
  select * into v from contract_links where token = p_token;

  -- Token inexistente, expirado e consumido respondem igual: quem estiver
  -- sondando nao aprende nada com a diferenca.
  if not found or v.status <> 'pendente' or v.expira_em <= now() then
    return jsonb_build_object('valido', false);
  end if;

  return jsonb_build_object(
    'valido',           true,
    'prefill_nome',     v.prefill_nome,
    'prefill_telefone', v.prefill_telefone,
    'plano_id',         v.plano_id
  );
end;
$$;

revoke all on function get_contract_link(text) from public;
grant execute on function get_contract_link(text) to anon, authenticated;

create or replace function submit_contratacao(p_token text, p_dados jsonb)
returns jsonb
language plpgsql security definer set search_path = public
as $$
declare
  v_link      contract_links%rowtype;
  v_tipo      pessoa_tipo;
  v_doc       text;
  v_plano     text;
  v_duplicado boolean;
begin
  -- for update serializa envios concorrentes no mesmo token: o segundo
  -- espera, encontra status 'preenchido' e e recusado.
  select * into v_link from contract_links where token = p_token for update;

  if not found or v_link.status <> 'pendente' or v_link.expira_em <= now() then
    raise exception 'link invalido' using errcode = 'P0001';
  end if;

  v_tipo := (p_dados ->> 'tipo')::pessoa_tipo;
  v_doc  := regexp_replace(coalesce(p_dados ->> 'documento', ''), '\D', '', 'g');

  if length(trim(coalesce(p_dados ->> 'nome_completo', ''))) < 3 then
    raise exception 'nome invalido' using errcode = 'P0002';
  end if;

  if v_tipo = 'fisica' and not validar_cpf(v_doc) then
    raise exception 'cpf invalido' using errcode = 'P0002';
  end if;

  if v_tipo = 'juridica' and not validar_cnpj(v_doc) then
    raise exception 'cnpj invalido' using errcode = 'P0002';
  end if;

  if coalesce(p_dados ->> 'email', '') !~ '^[^@\s]+@[^@\s]+\.[^@\s]+$' then
    raise exception 'email invalido' using errcode = 'P0002';
  end if;

  if length(regexp_replace(coalesce(p_dados ->> 'cep', ''), '\D', '', 'g')) <> 8 then
    raise exception 'cep invalido' using errcode = 'P0002';
  end if;

  if (p_dados ->> 'consentimento')::boolean is not true then
    raise exception 'consentimento obrigatorio' using errcode = 'P0002';
  end if;

  v_plano := coalesce(nullif(p_dados ->> 'plano_id', ''), v_link.plano_id);

  select exists (select 1 from clientes where documento = v_doc) into v_duplicado;

  insert into clientes (
    nome_completo, tipo, documento, rg, email, data_nascimento, telefone,
    cep, logradouro, numero, complemento, bairro, cidade, uf,
    plano_id, origem, link_id, criado_por, consentimento_em
  ) values (
    trim(p_dados ->> 'nome_completo'),
    v_tipo,
    v_doc,
    nullif(trim(coalesce(p_dados ->> 'rg', '')), ''),
    lower(trim(p_dados ->> 'email')),
    nullif(p_dados ->> 'data_nascimento', '')::date,
    regexp_replace(coalesce(p_dados ->> 'telefone', ''), '\D', '', 'g'),
    regexp_replace(coalesce(p_dados ->> 'cep', ''), '\D', '', 'g'),
    trim(p_dados ->> 'logradouro'),
    trim(p_dados ->> 'numero'),
    nullif(trim(coalesce(p_dados ->> 'complemento', '')), ''),
    trim(p_dados ->> 'bairro'),
    trim(p_dados ->> 'cidade'),
    upper(trim(p_dados ->> 'uf')),
    v_plano,
    'link_vendedor',
    v_link.id,
    v_link.gerado_por,
    now()
  );

  update contract_links
     set status = 'preenchido', consumido_em = now()
   where id = v_link.id;

  -- Devolve o minimo: nunca id, nunca dado do cliente.
  return jsonb_build_object('ok', true, 'duplicado', v_duplicado);
end;
$$;

revoke all on function submit_contratacao(text, jsonb) from public;
grant execute on function submit_contratacao(text, jsonb) to anon, authenticated;
```

```sql
-- agendamento automatico ainda. Cliente instalado nao e tocado: fica

         cidade          = '',
```

```ts
export type AuthStep = 'carregando' | 'login' | 'mfa-enroll' | 'mfa-challenge' | 'pronto'

export interface AuthStepInput {
  loading: boolean
  hasSession: boolean
  /** Fatores TOTP já verificados na conta. */
  verifiedFactors: number
  /** Nível de garantia da sessão atual, como o Supabase informa. */
  aal: 'aal1' | 'aal2' | null
}

/**
 * Decide em que etapa da autenticação o usuário está.
 *
 * Separada do hook de propósito: é a única parte com regra de verdade, e
 * assim ela é testável sem montar componente nem falsificar o SDK.
 */
export function resolveAuthStep(input: AuthStepInput): AuthStep {
  if (input.loading) {
    return 'carregando'
  }
  if (!input.hasSession) {
    return 'login'
  }
  if (input.verifiedFactors === 0) {
    return 'mfa-enroll'
  }
  if (input.aal !== 'aal2') {
    return 'mfa-challenge'
  }
  return 'pronto'
}
```

```ts
import { useCallback, useEffect, useState } from 'react'
import { supabase } from '@/shared/lib/supabase'
import { resolveAuthStep, type AuthStep } from './authStep'

interface SessionState {
  loading: boolean
  hasSession: boolean
  verifiedFactors: number
  aal: 'aal1' | 'aal2' | null
  userId: string | null
  email: string | null
}

const INITIAL: SessionState = {
  loading: true,
  hasSession: false,
  verifiedFactors: 0,
  aal: null,
  userId: null,
  email: null,
}

export function useSession() {
  const [state, setState] = useState<SessionState>(INITIAL)

  const load = useCallback(async () => {
    const { data: sessionData } = await supabase.auth.getSession()
    const session = sessionData.session

    if (!session) {
      setState({ ...INITIAL, loading: false })
      return
    }

    const { data: factors } = await supabase.auth.mfa.listFactors()
    const { data: aal } = await supabase.auth.mfa.getAuthenticatorAssuranceLevel()

    setState({
      loading: false,
      hasSession: true,
      verifiedFactors: factors?.totp.length ?? 0,
      aal: (aal?.currentLevel as 'aal1' | 'aal2' | null) ?? null,
      userId: session.user.id,
      email: session.user.email ?? null,
    })
  }, [])

  useEffect(() => {
    void load()
    const { data } = supabase.auth.onAuthStateChange(() => {
      void load()
    })
    return () => data.subscription.unsubscribe()
  }, [load])

  const sair = useCallback(async () => {
    await supabase.auth.signOut()
    setState({ ...INITIAL, loading: false })
  }, [])

  const step: AuthStep = resolveAuthStep(state)

  return { step, userId: state.userId, email: state.email, refresh: load, sair }
}
```

```tsx
}
```

```ts
  const [gerando, setGerando] = useState(false)
```

```ts
import { useCallback, useEffect, useState } from 'react'
import { supabase } from '@/shared/lib/supabase'
import { STATUS_ORDER, type ClienteStatus } from './statusLabels'

export interface ClienteResumo {
  id: string
  nome_completo: string
  documento: string
  telefone: string
  plano_id: string
  status: ClienteStatus
  criado_em: string
}

export interface FiltroClientes {
  /** Nome, telefone ou documento. Só dígitos são comparados nos dois últimos. */
  busca: string
  status: ClienteStatus | 'todos'
}

const COLUNAS = 'id, nome_completo, documento, telefone, plano_id, status, criado_em'

export function useClientes(filtro: FiltroClientes) {
  const [clientes, setClientes] = useState<ClienteResumo[]>([])
  const [carregando, setCarregando] = useState(true)
  const [erro, setErro] = useState<string | null>(null)

  const carregar = useCallback(async () => {
    setCarregando(true)
    setErro(null)

    let query = supabase.from('clientes').select(COLUNAS).order('criado_em', { ascending: false })

    if (filtro.status !== 'todos') {
      query = query.eq('status', filtro.status)
    }

    const termo = filtro.busca.trim()
    if (termo) {
      const digitos = termo.replace(/\D/g, '')
      const alvos = [`nome_completo.ilike.%${termo}%`]
      if (digitos) {
        alvos.push(`documento.ilike.%${digitos}%`, `telefone.ilike.%${digitos}%`)
      }
      query = query.or(alvos.join(','))
    }

    const { data, error } = await query

    setCarregando(false)

    if (error) {
      setErro('Não foi possível carregar a lista.')
      return
    }

    setClientes((data ?? []) as ClienteResumo[])
  }, [filtro.busca, filtro.status])

  useEffect(() => {
    void carregar()
  }, [carregar])

  const contadores = STATUS_ORDER.reduce(
    (acc, status) => {
      acc[status] = clientes.filter((cliente) => cliente.status === status).length
      return acc
    },
    {} as Record<ClienteStatus, number>,
  )

  return { clientes, contadores, carregando, erro, recarregar: carregar }
}
```

Os contadores refletem o resultado filtrado. Isso é intencional: eles somam o que está na tela.

```ts
/**
 * Validação de formulário. Roda no navegador por conveniência — o que
 * realmente barra dado ruim é a mesma checagem dentro de
 * `submit_contratacao`, no Postgres.
 */

const somenteDigitos = (value: string) => value.replace(/\D/g, '')

function digitoValido(digits: string, ate: number, pesoInicial: number): boolean {
  let soma = 0
  for (let i = 0; i < ate; i += 1) {
    soma += Number(digits[i]) * (pesoInicial - i)
  }
  let resto = (soma * 10) % 11
  if (resto === 10) {
    resto = 0
  }
  return resto === Number(digits[ate])
}

export function isValidCpf(value: string): boolean {
  const d = somenteDigitos(value)
  if (d.length !== 11 || /^(\d)\1{10}$/.test(d)) {
    return false
  }
  return digitoValido(d, 9, 10) && digitoValido(d, 10, 11)
}

export function isValidCnpj(value: string): boolean {
  const d = somenteDigitos(value)
  if (d.length !== 14 || /^(\d)\1{13}$/.test(d)) {
    return false
  }

  const checar = (ate: number, pesos: number[]) => {
    let soma = 0
    for (let i = 0; i < ate; i += 1) {
      soma += Number(d[i]) * pesos[i]
    }
    const resto = soma % 11
    return (resto < 2 ? 0 : 11 - resto) === Number(d[ate])
  }

  return (
    checar(12, [5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2]) &&
    checar(13, [6, 5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2])
  )
}

export function isValidEmail(value: string): boolean {
  return /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(value.trim())
}

export function isValidCep(value: string): boolean {
  return somenteDigitos(value).length === 8
}

/** Espera uma data ISO (`AAAA-MM-DD`). Contrato exige titular maior de idade. */
export function isAdult(isoDate: string): boolean {
  const nascimento = new Date(`${isoDate}T00:00:00`)
  if (Number.isNaN(nascimento.getTime())) {
    return false
  }

  const limite = new Date()
  limite.setFullYear(limite.getFullYear() - 18)
  return nascimento <= limite
}
```

```ts
export type MaskKind = 'tel' | 'cep' | 'cpf' | 'cnpj' | 'date'

/** Insere separadores conforme o usuário digita, sem antecipá-los. */
function agrupar(digits: string, tamanhos: number[], separadores: string[]): string {
  let resto = digits
  let saida = ''

  for (let i = 0; i < tamanhos.length && resto.length > 0; i += 1) {
    const pedaco = resto.slice(0, tamanhos[i])
    saida += (i === 0 ? '' : separadores[i - 1]) + pedaco
    resto = resto.slice(tamanhos[i])
  }

  return saida
}

const MASKS: Record<MaskKind, (digits: string) => string> = {
  // ...tel e cep permanecem exatamente como estão...
  cpf: (digits) => agrupar(digits.slice(0, 11), [3, 3, 3, 2], ['.', '.', '-']),
  cnpj: (digits) => agrupar(digits.slice(0, 14), [2, 3, 3, 4, 2], ['.', '.', '/', '-']),
  date: (digits) => agrupar(digits.slice(0, 8), [2, 2, 4], ['/', '/']),
}

/** Exportada para teste: aplica a máscara a uma string já sem pontuação. */
export function applyMask(kind: MaskKind, raw: string): string {
  return MASKS[kind](raw.replace(/\D/g, ''))
```

```ts
/**
 * Chamada de RPC do Supabase por `fetch` puro.
 *
 * O site não instala `@supabase/supabase-js` de propósito: só existem duas
 * chamadas, nenhuma sessão para gerenciar, e o SDK custaria cerca de 35 kB
 * gzip num bundle afinado para 98 de Lighthouse.
 *
 * A chave anon é pública por natureza. Quem protege os dados é o RLS: esse
 * papel não tem acesso a nenhuma tabela, apenas permissão de executar duas
 * funções que nunca devolvem dado de cliente.
 */
const URL_BASE = import.meta.env.VITE_SUPABASE_URL
const ANON_KEY = import.meta.env.VITE_SUPABASE_ANON_KEY

export class RpcError extends Error {
  readonly code: string | null

  constructor(message: string, code: string | null) {
    super(message)
    this.name = 'RpcError'
    this.code = code
  }
}

export async function rpc<T>(fn: string, args: Record<string, unknown>): Promise<T> {
  const response = await fetch(`${URL_BASE}/rest/v1/rpc/${fn}`, {
    method: 'POST',
    headers: {
      apikey: ANON_KEY,
      Authorization: `Bearer ${ANON_KEY}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify(args),
  })

  if (!response.ok) {
    const corpo: { message?: string; code?: string } | null = await response
      .json()
      .catch(() => null)
    throw new RpcError(corpo?.message ?? `Falha em ${fn}`, corpo?.code ?? null)
  }

  return (await response.json()) as T
}
```

```ts
import { useEffect, useState } from 'react'
import { rpc } from '@/shared/lib/supabaseRest'
import type { ContratacaoLink } from './types'

interface RespostaBruta {
  valido: boolean
  prefill_nome: string | null
  prefill_telefone: string | null
  plano_id: string
}

export function useContratacaoLink(token: string | undefined) {
  const [link, setLink] = useState<ContratacaoLink | null>(null)
  const [carregando, setCarregando] = useState(true)
  const [falhaDeRede, setFalhaDeRede] = useState(false)

  useEffect(() => {
    if (!token) {
      setLink({ valido: false })
      setCarregando(false)
      return
    }

    let ativo = true

    rpc<RespostaBruta>('get_contract_link', { p_token: token })
      .then((resposta) => {
        if (!ativo) return
        setLink(
          resposta.valido
            ? {
                valido: true,
                prefillNome: resposta.prefill_nome,
                prefillTelefone: resposta.prefill_telefone,
                planoId: resposta.plano_id,
              }
            : { valido: false },
        )
      })
      .catch(() => {
        // Falha de rede não é o mesmo que link ruim: mandar o cliente para a
        // tela de "link inválido" por causa de Wi-Fi caindo seria mentira.
        if (ativo) setFalhaDeRede(true)
      })
      .finally(() => {
        if (ativo) setCarregando(false)
      })

    return () => {
      ativo = false
    }
  }, [token])

  return { link, carregando, falhaDeRede }
}
```

```ts
        // tentar de novo sem redigitar tudo.
        setErro(
          causa instanceof RpcError && causa.code && MENSAGEM_POR_CODIGO[causa.code]
            ? MENSAGEM_POR_CODIGO[causa.code]
            : 'Não conseguimos enviar agora. Verifique sua conexão e tente de novo.',
        )
      } finally {
        setEnviando(false)
      }
    },
    [token],
  )

  return { enviar, enviando, erro, concluido }
}
```
